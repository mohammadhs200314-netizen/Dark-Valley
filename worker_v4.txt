const enc = new TextEncoder();
const id = () => crypto.randomUUID();
const hex = b => [...new Uint8Array(b)].map(x => x.toString(16).padStart(2, "0")).join("");
async function hash(s) { return hex(await crypto.subtle.digest("SHA-256", enc.encode(s))); }
async function passHash(password, salt) {
  const key = await crypto.subtle.importKey("raw", enc.encode(password), "PBKDF2", false, ["deriveBits"]);
  return hex(await crypto.subtle.deriveBits({ name: "PBKDF2", salt: enc.encode(salt), iterations: 180000, hash: "SHA-256" }, key, 256));
}
function cookie(name, value, maxAge) { return `${name}=${value}; HttpOnly; Secure; SameSite=Lax; Path=/; Max-Age=${maxAge}`; }
function getCookie(req, name) {
  return (req.headers.get("Cookie") || "").split(";").map(x => x.trim()).find(x => x.startsWith(name + "="))?.slice(name.length + 1);
}
function corsHeaders(req, env) {
  const origin = req.headers.get("Origin");
  const allowed = String(env.ALLOWED_ORIGINS || "").split(",").map(x => x.trim()).filter(Boolean);
  const sameOrigin = origin && origin === new URL(req.url).origin;
  const ok = !origin || sameOrigin || allowed.includes(origin);
  const h = new Headers({
    "Vary": "Origin",
    "Access-Control-Allow-Credentials": "true",
    "Access-Control-Allow-Headers": "Content-Type, Range",
    "Access-Control-Expose-Headers": "Content-Length, Content-Range, Accept-Ranges",
    "Access-Control-Allow-Methods": "GET,POST,DELETE,OPTIONS",
    "Access-Control-Max-Age": "86400",
    "X-Content-Type-Options": "nosniff",
    "Referrer-Policy": "strict-origin-when-cross-origin",
    "X-Frame-Options": "DENY",
    "Permissions-Policy": "camera=(), microphone=(), geolocation=()"
  });
  if (origin && ok) h.set("Access-Control-Allow-Origin", origin);
  return { headers: h, ok };
}
const json = (data, status = 200, headers = new Headers()) => {
  const h = new Headers(headers); h.set("Content-Type", "application/json; charset=utf-8");
  return new Response(JSON.stringify(data), { status, headers: h });
};
async function body(req) { try { return await req.json(); } catch { return {}; } }
async function auth(req, env) {
  const token = getCookie(req, "dv_session"); if (!token) return null;
  const tokenHash = await hash(token);
  return await env.DB.prepare(
    "SELECT u.id,u.name,u.email,u.role FROM sessions s JOIN users u ON u.id=s.user_id WHERE s.token_hash=? AND s.expires_at>?"
  ).bind(tokenHash, Date.now()).first();
}
const isAdmin = u => !!u && u.role === "admin";
async function makeSession(uid, env, headers) {
  const token = hex(crypto.getRandomValues(new Uint8Array(32)));
  await env.DB.prepare("INSERT INTO sessions(token_hash,user_id,expires_at) VALUES(?,?,?)")
    .bind(await hash(token), uid, Date.now() + 30 * 86400000).run();
  const user = await env.DB.prepare("SELECT id,name,email,role FROM users WHERE id=?").bind(uid).first();
  const h = new Headers(headers); h.append("Set-Cookie", cookie("dv_session", token, 30 * 86400));
  return json({ user }, 200, h);
}
function quotaBytes(env) {
  const mb = Math.max(10, Math.min(2048, Number(env.USER_UPLOAD_QUOTA_MB || 100)));
  return Math.floor(mb * 1024 * 1024);
}
function bytesAllowed(file, max, label) {
  if (!(file instanceof File) || file.size <= 0) throw new Error(`فایل ${label} انتخاب نشده است`);
  if (file.size > max) throw new Error(`حجم ${label} از حد مجاز بیشتر است`);
}

// Best-effort isolate-local limiter. Configure Cloudflare Rate Limiting/WAF for global enforcement.
const loginAttempts = new Map();
function loginLimited(key) {
  const now = Date.now(), old = loginAttempts.get(key);
  if (!old || now - old.startedAt > 10 * 60 * 1000) { loginAttempts.set(key, { startedAt: now, count: 1 }); return false; }
  old.count++; return old.count > 10;
}
async function userQuota(env, userId) {
  const used = await env.DB.prepare(
    "SELECT COALESCE(SUM(storage_bytes),0) AS used FROM submissions WHERE user_id=? AND status IN ('pending','approved')"
  ).bind(userId).first();
  const recent = await env.DB.prepare(
    "SELECT COUNT(*) AS n FROM submissions WHERE user_id=? AND submitted_at>?"
  ).bind(userId, Date.now() - 86400000).first();
  const limit = quotaBytes(env), usedBytes = Number(used?.used || 0);
  return { usedBytes, quotaBytes: limit, remainingBytes: Math.max(0, limit - usedBytes), remainingUploads: Math.max(0, 2 - Number(recent?.n || 0)) };
}
async function removeSubmissionObjects(env, row) {
  if (row?.audio_key) await env.AUDIO.delete(row.audio_key);
  if (row?.cover_key) await env.AUDIO.delete(row.cover_key);
}

        if (!txt || txt.length > 1200) return json({ error: "متن دیدگاه نامعتبر است" }, 400, headers);
        await env.DB.prepare("INSERT INTO comments(id,song_id,user_id,body,created_at) VALUES(?,?,?,?,?)").bind(id(), String(b.songId), u.id, txt, Date.now()).run();
        return json({ ok: true }, 201, headers);
      }
      if (p === "/api/tickets" && method === "POST") {
        if (!u) return json({ error: "برای ارسال تیکت وارد حساب شو" }, 401, headers);
        const b = await body(req), sub = String(b.subject || "").trim(), txt = String(b.body || "").trim();
        if (!sub || !txt || sub.length > 160 || txt.length > 3000) return json({ error: "موضوع و متن تیکت لازم است" }, 400, headers);
        await env.DB.prepare("INSERT INTO tickets(id,user_id,subject,body,status,created_at) VALUES(?,?,?,?,?,?)").bind(id(), u.id, sub, txt, "در انتظار پاسخ", Date.now()).run();
        return json({ ok: true }, 201, headers);
      }
      if (p === "/api/tickets" && method === "GET") {
        if (!u) return json({ error: "ابتدا وارد حساب شو" }, 401, headers);
        const q = isAdmin(u) ? "SELECT t.*,u.name FROM tickets t JOIN users u ON u.id=t.user_id ORDER BY t.created_at DESC" : "SELECT * FROM tickets WHERE user_id=? ORDER BY created_at DESC";
        const { results } = isAdmin(u) ? await env.DB.prepare(q).all() : await env.DB.prepare(q).bind(u.id).all();
        return json({ tickets: results }, 200, headers);
      }
      if (p === "/api/rooms" && method === "POST") {
        if (!u) return json({ error: "برای ساخت اتاق وارد حساب شو" }, 401, headers);
        const b = await body(req), code = "DV-" + crypto.randomUUID().slice(0,5).toUpperCase();
        const room = env.ROOMS.get(env.ROOMS.idFromName(code));
        await room.fetch("https://room/create", { method: "POST", body: JSON.stringify({ songId: b.songId || null, owner: u.id }) });
        return json({ code }, 201, headers);
      }
      const roomMatch = p.match(/^\/api\/rooms\/([^/]+)\/connect$/);
      if (roomMatch && method === "GET") {
        const room = env.ROOMS.get(env.ROOMS.idFromName(decodeURIComponent(roomMatch[1])));
        return room.fetch(req);
      }
      return json({ error: "مسیر پیدا نشد" }, 404, headers);
    } catch (e) {
      console.error("Dark Valley API error:", e);
      return json({ error: "خطای داخلی سرور رخ داد", requestId: crypto.randomUUID() }, 500, headers);
    }
  }
};
// Durable Object for synchronized listening rooms.
export class Room {
  constructor(state, env) { this.state = state; this.env = env; this.peers = new Set(); this.data = { songId: null, playing: false, time: 0 }; this.ready = this.state.storage.get("data").then(v => { if (v) this.data = v; }); }
  async fetch(req) {
    await this.ready; const u = new URL(req.url);
    if (u.pathname === "/create") { this.data = await req.json(); await this.state.storage.put("data", this.data); return new Response("ok"); }
    if (u.pathname === "/connect") {
      if (req.headers.get("Upgrade") !== "websocket") return new Response("WebSocket required", { status: 426 });
      const pair = new WebSocketPair(), client = pair[0], server = pair[1]; server.accept(); this.peers.add(server);
      server.send(JSON.stringify({ type: "state", ...this.data }));
      server.addEventListener("message", async e => {
        let m; try { m = JSON.parse(e.data); } catch { return; }
        if (m.type === "play") { this.data = { ...this.data, playing: true, songId: m.songId || this.data.songId, time: 0 }; }
        if (m.type === "pause") this.data.playing = false;
        if (m.type === "seek") this.data.time = Number(m.time) || 0;
        await this.state.storage.put("data", this.data);
        for (const peer of this.peers) { try { peer.send(JSON.stringify({ ...m, ...this.data })); } catch { this.peers.delete(peer); } }
      });
      server.addEventListener("close", () => this.peers.delete(server));
      return new Response(null, { status: 101, webSocket: client });
    }
    return new Response("Not found", { status: 404 });
  }
}
